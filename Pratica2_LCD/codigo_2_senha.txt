delay:
	mov r7, #010h
here:
	djnz r7, here
ret


delay_longo:
    mov r7, #0ffh
    mov B, #0ffh

	djnz r7, again
again:
    djnz B, here2

ret

	setB p0.1
	JB KeyPressFlag, Done

	JB KeyPressFlag, Done

	jmp ScanKey
ret

Done:

PrintString:
	clr A
	movc A, @A+DPTR  
	jz Print_Done	 
	mov R2, A    	 
	setb RS      	 
	call sendchar
	inc DPTR      	; Próximo caractere
	sjmp PrintString
Print_Done:
	ret
